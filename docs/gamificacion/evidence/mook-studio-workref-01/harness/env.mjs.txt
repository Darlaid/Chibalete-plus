// Entorno aislado: server.js real + Vite dev real, solo con stores temporales.
import { spawn } from 'node:child_process';
import crypto from 'node:crypto';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { pathToFileURL } from 'node:url';

export const REPO = 'D:/001 - app - Chibalete+';
export const API = 'http://127.0.0.1:3000';
export const WEB = 'http://localhost:5173';
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

export const CATALOG = [
    { id: 'obra-a', titulo: 'Obra Alfa', autor: 'Sintética', tipo: 'libro', status: 'disponible' },
    { id: 'obra-b', titulo: 'Obra Beta', autor: 'Sintética', tipo: 'libro_album', status: 'disponible' },
    { id: 'obra-v', titulo: 'Obra Video', autor: 'Sintética', tipo: 'video', status: 'disponible' },
    { id: 'coleccion', titulo: 'NE Coleccion', autor: 'x', tipo: 'libro', status: 'disponible', isCollection: true },
    { id: 'capitulo', titulo: 'NE Capitulo', autor: 'x', tipo: 'libro', status: 'disponible', parentId: 'coleccion' },
    { id: 'guia', titulo: 'NE Guia', autor: 'x', tipo: 'guia', status: 'disponible', parentId: 'obra-a' },
    { id: 'podcast', titulo: 'NE Podcast', autor: 'x', tipo: 'podcast', status: 'disponible' },
    { id: 'articulo', titulo: 'NE Articulo', autor: 'x', tipo: 'articulo_pedagogico', status: 'disponible' },
    { id: 'pieza', titulo: 'NE Pieza', autor: 'x', tipo: 'libro', status: 'disponible', standalone: false },
    { id: 'procesando', titulo: 'NE Procesando', autor: 'x', tipo: 'libro', status: 'procesando' },
];
export const ELIGIBLE = ['Obra Alfa', 'Obra Beta', 'Obra Video'];
export const INELIGIBLE = CATALOG.filter((c) => c.titulo.startsWith('NE ')).map((c) => c.titulo);

const reading = (id, ref) => ({ id, type: 'READING', title: `Leer ${id}`, required: true, resourceRef: ref, config: {} });
const production = (id, ref) => ({ id, type: 'PRODUCTION', title: `Producir ${id}`, required: true, ...(ref ? { resourceRef: ref } : {}), config: { consigna: 'Escribe', criterioRevision: 'Claridad', minPalabras: 3, maxPalabras: 30 } });
const mods = (...nodes) => [{ id: 'm1', title: 'Módulo uno', nodes }];

export async function up() {
    const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'chp_studio_e2e_'));
    const data = path.join(dir, 'data'), sqlite = path.join(dir, 'sqlite');
    fs.mkdirSync(data); fs.mkdirSync(sqlite); fs.mkdirSync(path.join(dir, 'uploads'));
    const contentFile = path.join(data, 'content.json');
    const mookFile = path.join(data, 'mook_db.json');
    fs.writeFileSync(contentFile, JSON.stringify(CATALOG));
    for (const f of ['access_db.json', 'progress_db.json', 'groups_db.json', 'schools_db.json', 'school_configs.json', 'leo_evidence_db.json']) fs.writeFileSync(path.join(data, f), '[]');
    const user = (id, roles) => ({ id, email: `${id}@t.test`, nombre_completo: id, roles, accountStatus: 'active', password: 'x' });
    fs.writeFileSync(path.join(dir, 'users.json'), JSON.stringify([user('u-admin', ['administrador']), user('u-lector', ['lector'])]));

    // Borradores sintéticos. E2E Legado simula un borrador anterior al contrato.
    const Store = await import(pathToFileURL(path.join(REPO, 'server/lib/experienceStore.js')).href);
    const doc = Store.emptyMookStore();
    const seed = (slug, title, nodes) => { const e = Store.createExperience(doc, { slug, title }); const v = Store.createDraftVersion(doc, e.id, { modules: mods(...nodes) }); return { exp: e.id, ver: v.id }; };
    const ids = {
        flujo: seed('e2e-flujo', 'E2E Flujo', [reading('r1', 'obra-a'), production('p1')]),
        legado: seed('e2e-legado', 'E2E Legado', [reading('r1', 'podcast'), production('p1', 'podcast')]),
        retiro: seed('e2e-retiro', 'E2E Retiro', [reading('r1', 'obra-b'), production('p1', 'obra-b')]),
        cerrado: seed('e2e-cerrado', 'E2E Cerrado', [reading('r1', 'obra-a'), production('p1')]),
    };
    fs.writeFileSync(mookFile, JSON.stringify(doc));

    const db = (n) => path.join(sqlite, n);
    const gam = path.join(dir, 'gam', 'gamification.db');
    const env = { ...process.env, NODE_ENV: 'test', SESSION_AUTH_MODE: 'off', ACCESS_FALLBACK_MODE: 'restricted', PORT: '3000', USERS_DB: path.join(dir, 'users.json'), CHP_DATA_DIR: data, UPLOADS_ROOT: path.join(dir, 'uploads'),
        ACCESS_DB: path.join(data, 'access_db.json'), GAMIFICATION_ENABLED: '', XP_EARNING_ENABLED: '', PM_EARNING_ENABLED: '', GAMIFICATION_DB: gam, GAMIFICATION_MOOK_SINCE: '', EXPERIENCE_EVENTS_BACKBONE_ENABLED: '',
        EVENTS_SQLITE_PATH: db('events.db'), ARCHIVE_SQLITE_PATH: db('events.archive.db'), INSIGHTS_SQLITE_PATH: db('insights.db'), PROGRESS_SQLITE_PATH: db('progress.db'), OFFLINE_ASSIGNMENT_DB_PATH: db('offline.db'),
        ROYALTIES_DB: db('royalties.db'), IDENTITY_DB: db('identity.db'), SESSIONS_DB: db('sessions.db'), GROUPS_DB: path.join(data, 'groups_db.json'), SCHOOLS_DB: path.join(data, 'schools_db.json'), LEO_EVIDENCE_DB: path.join(data, 'leo_evidence_db.json') };
    let logs = '';
    const api = spawn(process.execPath, [path.join(REPO, 'server/server.js')], { cwd: dir, env, stdio: ['ignore', 'pipe', 'pipe'] });
    api.stdout.on('data', (d) => { logs += d; }); api.stderr.on('data', (d) => { logs += d; });
    const webEnv = { ...process.env }; delete webEnv.NODE_ENV;
    const web = spawn(process.execPath, [path.join(REPO, 'node_modules/vite/bin/vite.js'), '--port', '5173', '--strictPort'], { cwd: REPO, env: webEnv, stdio: ['ignore', 'pipe', 'pipe'] });
    web.stdout.on('data', (d) => { logs += d; }); web.stderr.on('data', (d) => { logs += d; });
    for (let i = 0; ; i++) {
        try { if ((await fetch(`${API}/api/health`)).ok && (await fetch(WEB)).ok) break; } catch { /* arrancando */ }
        if (i > 120) throw new Error(`no arrancó:\n${logs.slice(-2000)}`);
        await sleep(500);
    }
    const stored = () => JSON.parse(fs.readFileSync(mookFile, 'utf8'));
    const call = async (method, url, body) => { const r = await fetch(`${API}${url}`, { method, headers: { 'content-type': 'application/json', 'x-user-id': 'u-admin' }, body: body ? JSON.stringify(body) : undefined }); return { status: r.status, body: await r.json().catch(() => null) }; };
    const down = async () => {
        for (const c of [api, web]) { c.kill(); await new Promise((r) => { c.on('close', r); setTimeout(r, 3000); }); }
        fs.rmSync(dir, { recursive: true, force: true, maxRetries: 5 });
    };
    return { dir, ids, contentFile, stored, call, down, gam, logs: () => logs, mods, reading, production };
}
